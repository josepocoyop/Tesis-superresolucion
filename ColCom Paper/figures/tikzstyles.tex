  },
}

% ancho de las vinetas de los diagramas
\newcommand{\thumbw}{0.95cm}

% Fotograma suelto: \singleframe{nombre}{archivo}{posicion}
\newcommand{\singleframe}[3]{%
  \node[photo] (#1) at #3 {\includegraphics[width=\thumbw]{#2}};%
}

% Pila de fotogramas: \framestack{nombre}{archivo}{posicion}
% El fotograma va al frente y detras se dibujan dos hojas desplazadas.
\newcommand{\framestack}[3]{%
  \node[photo] (#1) at #3 {\includegraphics[width=\thumbw]{#2}};%
  \begin{scope}[on background layer]
    \draw[fill=white, draw=black!65, line width=0.4pt]
      ($(#1.south west)+(0.06,0.06)$) rectangle ($(#1.north east)+(0.06,0.06)$);
    \draw[fill=white, draw=black!65, line width=0.4pt]
      ($(#1.south west)+(0.12,0.12)$) rectangle ($(#1.north east)+(0.12,0.12)$);
  \end{scope}%
}
